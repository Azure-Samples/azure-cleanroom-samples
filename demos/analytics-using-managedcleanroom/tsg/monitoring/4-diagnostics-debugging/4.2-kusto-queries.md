# 4.2 Kusto Queries Library

## Overview

Ready-to-use Kusto queries for debugging, monitoring, and reporting. These queries run on the Geneva → Kusto pipeline tables.

> **Note:** Table names and schemas below are based on the proposed structured logging schema (Section 1.3.2). Exact names will be finalized when the pipeline is fully operational.

## Tables

| Table | Source | Contents |
|-------|--------|----------|
| `CleanRoomOperations` | RP, Frontend, CM structured logs | Workflow tasks, operations, errors |
| `ACCRMetrics` | Geneva/MDM metrics | Custom app metrics (counters, histograms) |
| `ContainerLogV2` | Container Insights | Container stdout/stderr |
| `KubeEvents` | Container Insights | Pod lifecycle events |
| `AKSAudit` | AKS Diagnostic Settings | API server mutations |

## Debugging Queries

### Collaboration Lifecycle

```kql
// Full lifecycle of a specific collaboration
CleanRoomOperations
| where CollaborationName == "{name}"
| project Timestamp, WorkflowName, TaskName, Level, DurationMs, Message, ExceptionType
| order by Timestamp asc

// Collaboration creation waterfall (task durations)
CleanRoomOperations
| where CollaborationName == "{name}"
| where WorkflowName == "CreateCollaborationWorkflow"
| where DurationMs > 0
| project TaskName, DurationMs, DurationMin = round(DurationMs / 60000.0, 1), Level
| order by DurationMs desc
```

### Failure Investigation

```kql
// Failed operations in last 24h
CleanRoomOperations
| where Timestamp > ago(24h)
| where Level == "Error"
| summarize Count=count(), 
    FailureReasons=make_set(ExceptionType),
    Collaborations=make_set(CollaborationName)
  by TaskName, Region
| order by Count desc

// Specific failure details
CleanRoomOperations
| where Timestamp > ago(24h)
| where Level == "Error"
| where TaskName == "{taskName}"
| project Timestamp, CollaborationName, SubscriptionId, ExceptionType, ExceptionMessage, Message
| order by Timestamp desc

// MOBO failure rate by region
CleanRoomOperations
| where Timestamp > ago(7d)
| where TaskName == "CreateManagedResourceGroupTask"
| summarize Total=count(), Failed=countif(Level == "Error") by Region
| extend FailureRate = round(100.0 * Failed / Total, 1)
| order by FailureRate desc
```

### Performance Analysis

```kql
// Task duration percentiles over last 7 days
CleanRoomOperations
| where Timestamp > ago(7d)
| where DurationMs > 0
| summarize 
    p50=percentile(DurationMs, 50),
    p95=percentile(DurationMs, 95),
    p99=percentile(DurationMs, 99),
    Count=count()
  by TaskName
| extend p50_min=round(p50/60000.0,1), p95_min=round(p95/60000.0,1), p99_min=round(p99/60000.0,1)
| order by p95 desc

// Duration trend over time (daily)
CleanRoomOperations
| where Timestamp > ago(30d)
| where TaskName == "CreateClusterTask"
| where DurationMs > 0
| summarize P95=percentile(DurationMs, 95), Count=count() by bin(Timestamp, 1d)
| render timechart

// End-to-end collaboration creation time
CleanRoomOperations
| where Timestamp > ago(7d)
| where TaskName == "CreateCollaborationFinalize"
| where Level == "Information"
| summarize P50=percentile(DurationMs, 50), P95=percentile(DurationMs, 95) by Region
```

### Frontend Service

```kql
// Frontend error rate by endpoint
CleanRoomOperations
| where Timestamp > ago(1h)
| where Service == "Frontend"
| summarize Total=count(), Errors=countif(Level == "Error") by TaskName
| extend ErrorRate = round(100.0 * Errors / Total, 1)
| order by ErrorRate desc

// Auth failures breakdown
CleanRoomOperations
| where Timestamp > ago(24h)
| where Service == "Frontend"
| where TaskName contains "Auth"
| where Level == "Error"
| summarize Count=count() by ExceptionType, Message
| order by Count desc

// Query execution times
CleanRoomOperations
| where Timestamp > ago(7d)
| where Service == "Frontend"
| where TaskName == "QueryExecution"
| where DurationMs > 0
| summarize P50=percentile(DurationMs, 50), P95=percentile(DurationMs, 95), Count=count()
```

### Consortium Manager

```kql
// CCF health history
CleanRoomOperations
| where Timestamp > ago(24h)
| where Service == "ConsortiumManager"
| where TaskName contains "Health"
| project Timestamp, Level, Message
| order by Timestamp desc

// CCF recovery events
CleanRoomOperations
| where Timestamp > ago(7d)
| where Service == "ConsortiumManager"
| where TaskName contains "Recovery"
| project Timestamp, CollaborationName, Level, Message, DurationMs
| order by Timestamp desc
```

## Monitoring Queries (for Dashboards)

```kql
// Service health summary (last 1h)
CleanRoomOperations
| where Timestamp > ago(1h)
| summarize 
    Total=count(),
    Errors=countif(Level == "Error"),
    Warnings=countif(Level == "Warning")
  by Service
| extend ErrorRate = round(100.0 * Errors / Total, 1)

// Active collaborations by region
CleanRoomOperations
| where Timestamp > ago(1h)
| where TaskName == "HealthCheck"
| summarize dcount(CollaborationName) by Region

// Hourly operation volume
CleanRoomOperations
| where Timestamp > ago(24h)
| summarize Count=count() by bin(Timestamp, 1h), Service
| render timechart
```

## Customer-Specific Queries

```kql
// All operations for a specific customer
CleanRoomOperations
| where SubscriptionId == "{sub-id}"
| where Timestamp > ago(7d)
| project Timestamp, CollaborationName, WorkflowName, TaskName, Level, Message
| order by Timestamp desc

// Customer collaborations with health issues
CleanRoomOperations
| where Timestamp > ago(1h)
| where TaskName == "HealthCheck"
| where Level == "Error"
| project Timestamp, CollaborationName, SubscriptionId, TenantId, Message
| order by Timestamp desc
```
