## Example Report Queries

```kql
// Daily engineering health summary
let period = 24h;
let ops = CleanRoomOperations | where Timestamp > ago(period);
print
  TotalOperations = toscalar(ops | count),
  ErrorRate = toscalar(ops | summarize round(100.0 * countif(Level == "Error") / count(), 2)),
  CreateP95_min = toscalar(ops | where TaskName == "CreateCollaborationFinalize" | where DurationMs > 0 | summarize round(percentile(DurationMs, 95) / 60000.0, 1)),
  FailedCreates = toscalar(ops | where WorkflowName == "CreateCollaborationWorkflow" | where Level == "Error" | dcount(CollaborationName)),
  SuccessfulCreates = toscalar(ops | where TaskName == "CreateCollaborationFinalize" | where Level == "Information" | count()),
  Regions = toscalar(ops | summarize dcount(Region))

// Weekly failure trend (for regression detection)
CleanRoomOperations
| where Timestamp > ago(30d)
| summarize 
    Total=count(),
    Errors=countif(Level == "Error")
  by bin(Timestamp, 7d)
| extend ErrorRate = round(100.0 * Errors / Total, 2)
| render timechart
```
